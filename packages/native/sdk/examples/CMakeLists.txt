cmake_minimum_required(VERSION 3.20)
project(QuaNativeParent LANGUAGES C OBJC)
include("${CMAKE_CURRENT_LIST_DIR}/../qua-native.cmake")
add_executable(qua-native-protocol protocol.c)
target_link_libraries(qua-native-protocol PRIVATE QuaNative::Dynamic)
add_executable(qua-native-protocol-static protocol.c)
target_link_libraries(qua-native-protocol-static PRIVATE QuaNative::Static)
add_executable(qua-native-parent macos-parent.m)
target_compile_options(qua-native-parent PRIVATE -fobjc-arc)
target_link_libraries(qua-native-parent PRIVATE QuaNative::Dynamic "-framework Cocoa" "-framework QuartzCore")
